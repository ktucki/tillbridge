cmake_minimum_required(VERSION 3.14)
project(tillbridge VERSION 2.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(TILLBRIDGE_TESTS "Build the tests" ON)
option(TILLBRIDGE_OPENMP "Use OpenMP when available" ON)
option(TILLBRIDGE_EXAMPLES "Build the examples" ON)

file(GLOB TB_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp)
add_library(tillbridge_lib STATIC ${TB_SOURCES})
target_include_directories(tillbridge_lib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
set_target_properties(tillbridge_lib PROPERTIES OUTPUT_NAME tillbridge)
if(TILLBRIDGE_OPENMP)
  find_package(OpenMP)
  if(OpenMP_CXX_FOUND)
    target_link_libraries(tillbridge_lib PUBLIC OpenMP::OpenMP_CXX)
  endif()
endif()
if(MSVC)
  target_compile_options(tillbridge_lib PRIVATE /W3)
  target_compile_definitions(tillbridge_lib PUBLIC _CRT_SECURE_NO_WARNINGS _USE_MATH_DEFINES)
else()
  target_compile_options(tillbridge_lib PRIVATE -Wall -Wextra)
endif()

add_executable(tillbridge app/main.cpp)
target_link_libraries(tillbridge PRIVATE tillbridge_lib)
set_target_properties(tillbridge PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
foreach(cfg Release Debug RelWithDebInfo MinSizeRel)
  string(TOUPPER ${cfg} CFG)
  set_target_properties(tillbridge PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${CFG} ${CMAKE_BINARY_DIR})
endforeach()

if(TILLBRIDGE_EXAMPLES)
  add_executable(chisel_example examples/chisel/chisel_example.cpp)
  target_link_libraries(chisel_example PRIVATE tillbridge_lib)
  foreach(b oblique_impact fem_convergence verification)
    add_executable(${b} examples/benchmarks/${b}.cpp)
    target_link_libraries(${b} PRIVATE tillbridge_lib)
  endforeach()
  set_target_properties(chisel_example oblique_impact fem_convergence verification PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
endif()

if(TILLBRIDGE_TESTS)
  enable_testing()
  file(GLOB TB_TESTS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_*.cpp)
  add_executable(tillbridge_tests ${TB_TESTS})
  target_link_libraries(tillbridge_tests PRIVATE tillbridge_lib)
  target_compile_definitions(tillbridge_tests PRIVATE TB_TEST_DATA="${CMAKE_CURRENT_SOURCE_DIR}/tests/data")
  set_target_properties(tillbridge_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
  add_test(NAME unit COMMAND tillbridge_tests WORKING_DIRECTORY ${CMAKE_BINARY_DIR})
  add_test(NAME cli COMMAND ${CMAKE_COMMAND} -DTB=$<TARGET_FILE:tillbridge> -DDATA=${CMAKE_CURRENT_SOURCE_DIR}/tests/data
           -DWORK=${CMAKE_BINARY_DIR}/cli_smoke -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_smoke.cmake)
endif()

install(TARGETS tillbridge RUNTIME DESTINATION bin)
install(TARGETS tillbridge_lib ARCHIVE DESTINATION lib)
install(DIRECTORY include/tillbridge DESTINATION include)
